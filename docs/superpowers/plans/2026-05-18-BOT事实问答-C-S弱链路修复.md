# BOT 事实问答与 C/S 弱链路修复 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 BOT 对段位、分数、规则、漏分和 C/S 互动类问题基于真实事实回答，并修复 C/S 互动链路，使最后能用真实群消息复核分数并补录漏分。

**Architecture:** 新增问答意图层、运营事实层、轻量规则检索层，把运营事实问答从通用 LLM 中分离出来。C/S 弱链路新增可审计的 reaction/互动事件处理，确保事件接收、落库、计分、榜单维度、BOT 解释口径一致。

**Tech Stack:** TypeScript、Fastify、better-sqlite3、Vitest、飞书 OpenAPI/WS、现有 v2/v3 评分表。

```ts
export type BotQuestionIntentKind =
  | "level_status"
  | "score_breakdown"
  | "score_missing_check"
  | "cs_interaction_check"
  | "rules_query"
  | "course_or_homework_qa"
  | "general_chat";

export interface BotQuestionIntent {
  kind: BotQuestionIntentKind;
  reason: string;
}

function compactText(text: string): string {
  return text
    .replace(/@\S+/g, "")
    .replace(/\s+/g, "")
    .trim();
}

export function classifyBotQuestionIntent(rawText: string): BotQuestionIntent {
  const text = compactText(rawText);

  if (/C\/S|CS|点赞|表情|互动|互助|同伴|评论|反馈/.test(text)) {
    return { kind: "cs_interaction_check", reason: "interaction_keywords" };
  }
  if (/漏分|没算|未算|没加|补分|补录|查一下.*(消息|点赞|图片|作业|海报)/.test(text)) {
    return { kind: "score_missing_check", reason: "missing_score_keywords" };
  }
  if (/潜力股|段位|晋升|升级|升到|为什么.*(没|未|不|还).*升|如何.*(成为|晋升)/.test(text)) {
    return { kind: "level_status", reason: "level_keywords" };
  }
  if (/多少分|总分|维度分|排名|天梯榜/.test(text)) {
    return { kind: "score_breakdown", reason: "score_keywords" };
  }
  if (/规则|得分攻略|加分项目|prompt|提示词|上限|封顶|怎么算/.test(text)) {
    return { kind: "rules_query", reason: "rules_keywords" };
  }
  if (/作业|课程|上文|前面|刚才|RAG|图片怎么做|海报怎么做|解释一下/.test(text)) {
    return { kind: "course_or_homework_qa", reason: "course_context_keywords" };
  }
  return { kind: "general_chat", reason: "fallback" };
}
```

```ts
import type { LevelStatus } from "./chat-engine.js";

export interface ScoreFact {
  source: "v2" | "v3";
  categoryOrItem: string;
  dimension: "K" | "H" | "C" | "S" | "G";
  scoreDelta: number;
  status: string;
  decidedAt: string;
  note: string;
}

export interface InteractionFact {
  type: "reaction" | "peer_help" | "peer_review" | "mention";
  actorName: string;
  targetName: string;
  scoreDelta: number;
  status: string;
  occurredAt: string;
  note: string;
}

export interface BotFactServiceRepo {
  findMemberByOpenId(openId: string): { id: string; displayName: string; roleType: string; isParticipant: boolean; isExcludedFromBoard: boolean; currentLevel: number } | null;
  getLevelStatus(memberId: string): LevelStatus | null;
  listRecentScoreFacts(memberId: string, limit: number): ScoreFact[];
  listInteractionFacts(memberId: string, limit: number): InteractionFact[];
}

export type OperationalFacts =
  | { kind: "missing_member" }
  | { kind: "missing_status"; memberName: string }
  | { kind: "found"; memberId: string; memberName: string; status: LevelStatus; recentScores: ScoreFact[]; interactions: InteractionFact[] };

export function createBotFactService(input: { repo: BotFactServiceRepo }) {
  return {
    getOperationalFacts(args: { openId: string; question: string }): OperationalFacts {
      void args.question;
      const member = input.repo.findMemberByOpenId(args.openId);
      if (!member) return { kind: "missing_member" };
      const status = input.repo.getLevelStatus(member.id);
      if (!status) return { kind: "missing_status", memberName: member.displayName };
      return {
        kind: "found",
        memberId: member.id,
        memberName: member.displayName,
        status,
        recentScores: input.repo.listRecentScoreFacts(member.id, 8),
        interactions: input.repo.listInteractionFacts(member.id, 8),
      };
    },
  };
}
```

```ts
listRecentScoreFacts(memberId: string, limit: number): Array<{
  source: "v2" | "v3";
  categoryOrItem: string;
  dimension: "K" | "H" | "C" | "S" | "G";
  scoreDelta: number;
  status: string;
  decidedAt: string;
  note: string;
}> {
  const v2Rows = this.db.prepare(`
    SELECT 'v2' AS source, item_code AS categoryOrItem, dimension, score_delta AS scoreDelta,
           status, COALESCE(decided_at, created_at) AS decidedAt, COALESCE(review_note, source_ref, '') AS note
    FROM v2_scoring_item_events
    WHERE member_id = ?
    ORDER BY COALESCE(decided_at, created_at) DESC
    LIMIT ?
  `).all(memberId, limit) as Array<Record<string, unknown>>;

  const v3Rows = this.db.prepare(`
    SELECT 'v3' AS source, category AS categoryOrItem,
           CASE
             WHEN category IN ('ai_artifact', 'prompt_or_method') THEN 'C'
             WHEN category = 'peer_help' THEN 'S'
             WHEN category IN ('ai_practice_reflection', 'resource_recommendation') THEN 'G'
             WHEN category = 'formal_task' THEN 'H'
             ELSE 'K'
           END AS dimension,
           score_delta AS scoreDelta, status, decided_at AS decidedAt,
           COALESCE(review_note, rationale, '') AS note
    FROM ai_boot_score_events
    WHERE member_id = ?
    ORDER BY decided_at DESC
    LIMIT ?
  `).all(memberId, limit) as Array<Record<string, unknown>>;

  return [...v2Rows, ...v3Rows]
    .sort((a, b) => String(b.decidedAt).localeCompare(String(a.decidedAt)))
    .slice(0, limit)
    .map((row) => ({
      source: row.source as "v2" | "v3",
      categoryOrItem: String(row.categoryOrItem),
      dimension: row.dimension as "K" | "H" | "C" | "S" | "G",
      scoreDelta: Number(row.scoreDelta ?? 0),
      status: String(row.status),
      decidedAt: String(row.decidedAt),
      note: String(row.note ?? ""),
    }));
}
```

```ts
export interface RuleKnowledgeChunk {
  id: string;
  title: string;
  tags: string[];
  body: string;
}

export const RULE_KNOWLEDGE_CHUNKS: RuleKnowledgeChunk[] = [
  {
    id: "level-lv2",
    title: "Lv2 晋升路径",
    tags: ["段位", "晋升", "潜力股", "Lv2"],
    body: "Lv2 可以通过 24 分以上且 C/S/G 有有效贡献，或 32 分且单维达到 8 分，或 20 分以上、两个维度 5 分以上且 C/S/G 中有一个维度 5 分以上达成。",
  },
  {
    id: "cs-score",
    title: "C/S 互动得分",
    tags: ["C/S", "点赞", "互动", "互助", "peer_help"],
    body: "C 维度关注 AI 作品、方法、Prompt、创意应用和被认可的实践产物；S 维度关注同伴互助、有效反馈、互评和答疑。只 @BOT 问自己问题不能算 S。",
  },
  {
    id: "prompt-not-required",
    title: "Prompt 不是唯一硬条件",
    tags: ["prompt", "提示词", "规则", "作品"],
    body: "分享 Prompt 可以得分，但不是唯一硬条件。AI 图片、海报、客户演示、工作流、经验复盘、方法拆解等有效学习实践都可以进入评分。",
  },
];

export function retrieveRuleChunks(question: string, limit = 3): RuleKnowledgeChunk[] {
  const compact = question.toLowerCase();
  return RULE_KNOWLEDGE_CHUNKS
    .map((chunk) => ({
      chunk,
      score: chunk.tags.reduce((sum, tag) => sum + (compact.includes(tag.toLowerCase()) ? 1 : 0), 0),
    }))
    .filter((item) => item.score > 0)
    .sort((a, b) => b.score - a.score)
    .slice(0, limit)
    .map((item) => item.chunk);
}
```

`fact-answer.ts` 返回确定性回答：

```ts
import type { BotQuestionIntent } from "./intent-router.js";
import type { OperationalFacts } from "./fact-service.js";
import { retrieveRuleChunks } from "./rule-knowledge.js";

export interface FactAnswerInput {
  intent: BotQuestionIntent;
  facts: OperationalFacts;
  question: string;
}

export interface FactAnswerResult {
  handled: boolean;
  text: string;
}

export function buildFactAnswer(input: FactAnswerInput): FactAnswerResult {
  if (input.intent.kind === "general_chat" || input.intent.kind === "course_or_homework_qa") {
    return { handled: false, text: "" };
  }
  if (input.facts.kind === "missing_member") {
    return { handled: true, text: "我没有在天梯榜里找到你当前账号的数据。请让学员本人 @我，或说明要查哪位同学。" };
  }
  if (input.facts.kind === "missing_status") {
    return { handled: true, text: `@${input.facts.memberName} 我找到了你的账号，但当前段位/榜单数据不完整，需要运营侧核对。` };
  }

  const status = input.facts.status;
  const csg = `C${status.dimensions.C} / S${status.dimensions.S} / G${status.dimensions.G}`;
  if (input.intent.kind === "level_status") {
    return {
      handled: true,
      text: [
        `@${input.facts.memberName} 你现在是 Lv${status.currentLevel} ${status.currentLevelName}，总分 ${status.totalScore} 分${status.rank ? `，天梯榜第 ${status.rank} 名` : ""}。`,
        `当前维度：K${status.dimensions.K} / H${status.dimensions.H} / ${csg}。`,
        status.nextLevel ? `下一段位是 Lv${status.nextLevel} ${status.nextLevelName}。如果卡在潜力股，优先补 C/S/G：作品方法分享、同伴互助、真实复盘任意一类有效得分都能推动晋升。` : "你已经在当前最高段位。",
      ].join("\n"),
    };
  }

  if (input.intent.kind === "cs_interaction_check") {
    const rows = input.facts.interactions.filter((item) => item.scoreDelta > 0);
    if (rows.length === 0) {
      return { handled: true, text: `@${input.facts.memberName} 我暂时没有查到你已计入的 C/S 互动分。当前 ${csg}。如果你指的是某条点赞、互助或评论，请转发/引用那条消息，我会按消息记录核对。` };
    }
    return {
      handled: true,
      text: [`@${input.facts.memberName} 查到这些 C/S 互动记录：`, ...rows.slice(0, 5).map((row) => `- ${row.occurredAt.slice(0, 16)} ${row.type} +${row.scoreDelta}：${row.note}`)].join("\n"),
    };
  }

  const chunks = retrieveRuleChunks(input.question);
  return {
    handled: true,
    text: chunks.length > 0
      ? [`@${input.facts.memberName} 我按当前规则回答：`, ...chunks.map((chunk) => `- ${chunk.title}：${chunk.body}`)].join("\n")
      : `@${input.facts.memberName} 我能查到你当前总分 ${status.totalScore}，维度 K${status.dimensions.K} / H${status.dimensions.H} / ${csg}；这类规则问题我没有命中明确条款，需要运营侧补充规则文档。`,
  };
}
```

- [ ] **Step 2: 实现 normalizer**

```ts
export interface NormalizedReactionEvent {
  messageId: string;
  actorOpenId: string;
  emoji: string;
  occurredAt: string;
}

export function normalizeReactionEvent(data: unknown, now = () => new Date().toISOString()): NormalizedReactionEvent | null {
  const raw = data as { event?: Record<string, unknown> } & Record<string, unknown>;
  const event = (raw.event && typeof raw.event === "object" ? raw.event : raw) as Record<string, unknown>;
  const userId = event.user_id as { open_id?: string } | undefined;
  const operator = event.operator as { operator_id?: string } | undefined;
  const reactionType = event.reaction_type as { emoji_type?: string } | undefined;
  const messageId = typeof event.message_id === "string" ? event.message_id : "";
  const actorOpenId = userId?.open_id ?? operator?.operator_id ?? "";
  if (!messageId || !actorOpenId) return null;
  const createTime = typeof event.create_time === "string" ? Number(event.create_time) : NaN;
  return {
    messageId,
    actorOpenId,
    emoji: reactionType?.emoji_type ?? "",
    occurredAt: Number.isFinite(createTime) ? new Date(createTime).toISOString() : now(),
  };
}
```
